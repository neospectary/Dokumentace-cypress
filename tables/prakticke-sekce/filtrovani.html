Testování filtrování

Kompletní průvodce testováním filtrů v Cypressu. Zahrnuje aplikování filtrů, ověření aktivních filtrů a jejich odstraňování.

Otevření filtru

cy.dataCy('filters-button')
  .should('be.visible')
  .click();

cy.dataCy('modal-window')
  .should('be.visible')
  .within(() => {
    cy.dataCy('modal-title')
      .should('contain.text', 'Filtry');
  });

Aplikování filtru

Základní filtr

cy.dataCy('modal-window').within(() => {
  cy.fillInput('Email-container', 'test@example.com', {
    required: false
  });
  
  cy.intercept({ method: 'GET', url: '**/contact/grid/**' }).as('filter');
  cy.dataCy('save-button').click();
  cy.wait('@filter');
});

Více filtrů

cy.dataCy('modal-window').within(() => {
  cy.fillInput('Email-container', 'test@example.com');
  cy.fillInput('Name-container', 'Jméno');
  cy.selectFromList('Status', 0);
  
  cy.intercept({ method: 'GET', url: '**/contact/grid/**' }).as('filter');
  cy.dataCy('save-button').click();
  cy.wait('@filter');
});

Ověření aktivních filtrů

Zobrazení aktivních filtrů

cy.dataCy('active-filters-label')
  .should('be.visible')
  .and('contain.text', 'Aktivní filtry:');

cy.dataCy('Email-card').within(() => {
  cy.dataCy('filter-label')
    .should('be.visible')
    .and('contain.text', 'Email');
  cy.dataCy('filter-value')
    .should('be.visible')
    .and('contain.text', 'test@example.com');
});

Ověření filtrovaných výsledků

cy.dataCy('table-row').should('have.length', 3);
// Ověření, že všechny řádky obsahují filtrovanou hodnotu
cy.dataCy('table-row').each(($row) => {
  cy.wrap($row).should('contain.text', 'test@example.com');
});

Odstranění filtru

Odstranění jednoho filtru

cy.intercept({ method: 'GET', url: '**/contact/grid/**' }).as('removeFilter');
cy.dataCy('remove-filter-button')
  .should('be.visible')
  .click();
cy.wait('@removeFilter');

cy.dataCy('Email-card').should('not.exist');
cy.dataCy('active-filters-label').should('not.exist');

Vymazání všech filtrů v modálu

cy.dataCy('modal-window').within(() => {
  cy.dataCy('clear-filter-button')
    .find('button')
    .should('be.visible')
    .and('not.be.disabled')
    .click();
  
  cy.checkInputValue('Email-container', '');
  
  cy.intercept({ method: 'GET', url: '**/contact/grid/**' }).as('clearFilter');
  cy.dataCy('save-button').click();
  cy.wait('@clearFilter');
});

Filtr bez výsledků

cy.dataCy('modal-window').within(() => {

Kompletní příklad

describe('Filtrování kontaktů', () => {
  const EMAIL = 'cypress@smartemailing.cz';

  before(() => {
    cy.tokenAuth('/subscribers/contacts');
  });

  it('otevře modál filtrů', () => {
    cy.dataCy('filters-button').click();
    cy.dataCy('modal-window').should('be.visible');
  });

  it('aplikuje email filtr', () => {
    cy.dataCy('modal-window').within(() => {
      cy.fillInput('Email-container', EMAIL, { required: false });
      cy.intercept({ method: 'GET', url: '**/contact/grid/**' }).as('filter');
      cy.dataCy('save-button').click();
      cy.wait('@filter');
    });
  });

  it('zobrazí aktivní filtr', () => {
    cy.dataCy('active-filters-label').should('be.visible');
    cy.dataCy('Email-card').within(() => {
      cy.dataCy('filter-value').should('contain.text', EMAIL);
    });
  });

  it('zobrazí filtrované výsledky', () => {
    cy.dataCy('table-row').should('have.length', 3);
  });

  it('odstraní filtr', () => {
    cy.intercept({ method: 'GET', url: '**/contact/grid/**' }).as('removeFilter');
    cy.dataCy('remove-filter-button').click();
    cy.wait('@removeFilter');
    cy.dataCy('Email-card').should('not.exist');
  });
});
