Testování modálů

Kompletní průvodce testováním modálních oken v Cypressu. Zahrnuje otevírání, zavírání, vyplňování a potvrzování akcí v modálech.

Základní práce s modálem

Otevření modálu

cy.dataCy('open-modal-button').click();
cy.dataCy('modal-window').should('be.visible');

Zavření modálu

// Kliknutí na tlačítko Zavřít
cy.dataCy('modal-window').within(() => {
  cy.dataCy('close-button').click();
});
cy.dataCy('modal-window').should('not.exist');

// Nebo kliknutí mimo modál
cy.get('body').click(0, 0);
cy.dataCy('modal-window').should('not.exist');

Práce uvnitř modálu

Vyplňování formuláře v modálu

cy.dataCy('modal-window')
  .should('be.visible')
  .within(() => {
    cy.fillInput('name-container', 'Hodnota');
    cy.fillTextarea('description-container', 'Popis');
    cy.selectFromList('Hodnota', 0);
  });

Ověření obsahu modálu

cy.dataCy('modal-window')
  .should('be.visible')
  .within(() => {
    cy.dataCy('modal-title')
      .should('be.visible')
      .and('contain.text', 'Název modálu');
    
    cy.dataCy('modal-description')
      .should('be.visible')
      .and('contain.text', 'Popis modálu');
  });

Potvrzovací modály

Základní potvrzení

cy.dataCy('modal-window')
  .should('be.visible')
  .within(() => {
    cy.intercept({ method: 'DELETE', url: '**/api/item/**' }).as('delete');
    cy.dataCy('confirm-button')
      .should('be.visible')
      .and('contain.text', 'Smazat')
      .click();
    cy.wait('@delete');
  });
cy.dataCy('modal-window').should('not.exist');

Custom příkaz pro potvrzení (acceptModal)

// Definice
Cypress.Commands.add('acceptModal', (path, buttonName, { header, description, method = 'POST' } = {}) => {
  cy.intercept({ method: method, url: path }).as('acceptModal');
  cy.dataCy('modal-window')
    .should('be.visible')
    .within(() => {
      if (header) {
        cy.dataCy('modal-title').contains(header);
      }
      if (description) {
        cy.dataCy('modal-description').contains(description);
      }
      cy.dataCy('confirm-button').contains(buttonName).click();
    });
  cy.wait('@acceptModal');
});

// Použití
cy.acceptModal('**/api/item/**', 'Smazat', {
  header: 'Smazat záznam?',
  description: 'Opravdu si přejete smazat tento záznam?',
  method: 'DELETE'
});

Zrušení akce v modálu

cy.dataCy('modal-window')
  .should('be.visible')
  .within(() => {
    cy.dataCy('cancel-button').click();
  });
cy.dataCy('modal-window').should('not.exist');

Kompletní příklad

describe('Smazání záznamu', () => {
  before(() => {
    cy.tokenAuth('/items');
  });

  it('otevře modál pro smazání', () => {
    cy.dataCy('delete-button').click();
    cy.dataCy('modal-window')
      .should('be.visible')
      .within(() => {
        cy.dataCy('modal-title')
          .should('contain.text', 'Smazat záznam?');
      });
  });

  it('zobrazí správný obsah modálu', () => {
    cy.dataCy('modal-window').within(() => {
      cy.dataCy('modal-description')
        .should('contain.text', 'Opravdu si přejete smazat');
    });
  });

  it('potvrdí smazání', () => {
    cy.acceptModal('**/api/item/**', 'Smazat', {
      header: 'Smazat záznam?',
      method: 'DELETE'
    });
  });

  it('ověří, že modál byl zavřen', () => {
    cy.dataCy('modal-window').should('not.exist');
  });
});
