Testování tabulek

Kompletní průvodce testováním tabulek v Cypressu. Zahrnuje ověření řádků, sloupců, buněk, vyhledávání v tabulce a práci s daty.

Základní ověření tabulky

Ověření existence tabulky

cy.dataCy('table').should('be.visible');
cy.get('table').should('exist');

Ověření počtu řádků

// Přesný počet
cy.dataCy('table-row').should('have.length', 5);

// Minimální počet
cy.dataCy('table-row').should('have.length.greaterThan', 0);

// Maximální počet
cy.dataCy('table-row').should('have.length.at.most', 10);

Ověření počtu sloupců

cy.get('table thead th').should('have.length', 4);
cy.get('table tbody tr').first().find('td').should('have.length', 4);

Práce s konkrétními řádky

První řádek

cy.dataCy('table-row')
  .first()
  .within(() => {
    cy.dataCy('cell-name').should('contain.text', 'Název');
  });

Poslední řádek

cy.dataCy('table-row')
  .last()
  .within(() => {
    cy.dataCy('cell-name').should('contain.text', 'Poslední');
  });

Konkrétní řádek podle indexu

cy.dataCy('table-row')
  .eq(2) // třetí řádek (index 2)
  .within(() => {
    cy.dataCy('cell-email').should('contain.text', 'test@example.com');
  });

Řádek podle obsahu

cy.dataCy('table-row')
  .contains('Hledaný text')
  .within(() => {
    cy.dataCy('cell-name').should('be.visible');
  });

Ověření buněk

Ověření textu v buňce

cy.dataCy('table-row')
  .eq(0)
  .within(() => {
    cy.dataCy('cell-name').should('contain.text', 'Název');
    cy.dataCy('cell-email').should('contain.text', 'email@example.com');
  });

Ověření všech buněk v řádku

const EXPECTED_DATA = ['Název', 'Email', 'Telefon'];

cy.dataCy('table-row')
  .eq(0)
  .find('td')
  .each(($cell, index) => {
    cy.wrap($cell).should('contain.text', EXPECTED_DATA[index]);
  });

Komplexní ověření tabulky (Custom Command)

Vlastní příkaz pro ověření celé tabulky najednou: 

// Definice custom commandu
Cypress.Commands.add('checkPreviewTable', (data, { numberOfColumns = 1 } = {}) => {
  const ALL_ROWS = data.length;
  const ALL_COLUMNS = data.length * numberOfColumns;

  cy.dataCy('data-preview-table')
    .scrollIntoView()
    .within(() => {
      cy.get('tr').should('have.length', ALL_ROWS);
      cy.get('td').should('have.length', ALL_COLUMNS);

      for (let row = 0; row < ALL_ROWS; row++) {
        if (numberOfColumns === 1) {
          cy.get('tr')
            .eq(row)
            .find('td')
            .should('contain.text', data[row].join(', '));
        } else {
          for (let column = 0; column < numberOfColumns; column++) {
            cy.get('tr')
              .eq(row)
              .find('td')
              .eq(column)
              .should('contain.text', data[row][column]);
          }
        }
      }
    });
});

// Použití
const USERS = [
  ['jmeno1@domena.cz', 'Karel', 'Novák'],
  ['jmeno2@domena.cz', 'Honza', 'Mrkvička'],
];

cy.checkPreviewTable(USERS, { numberOfColumns: 3 });

Interakce s tabulkou

Kliknutí na řádek

cy.dataCy('table-row')
  .contains('Hledaný text')
  .click();

Kliknutí na tlačítko v řádku

cy.dataCy('table-row')
  .eq(0)
  .within(() => {
    cy.dataCy('detail-button').click();
  });

Checkbox v řádku

cy.dataCy('table-row')
  .eq(0)
  .find('input[type="checkbox"]')
  .check();

Práce s prázdnou tabulkou

// Ověření, že tabulka je prázdná
cy.dataCy('table-row').should('have.length', 0);

// Ověření zprávy o prázdné tabulce
cy.dataCy('empty-table')
  .should('be.visible')
  .and('contain.text', 'Žádné záznamy');

// Nebo
cy.dataCy('no-results-header')
  .should('be.visible')
  .and('contain.text', 'Nenalezeny žádné položky');

Ověření hlavičky tabulky

Kompletní příklad

describe('Tabulka kontaktů', () => {
  before(() => {
    cy.tokenAuth('/contacts');
  });

  it('zobrazí tabulku s kontakty', () => {
    cy.dataCy('table').should('be.visible');
    cy.dataCy('table-row').should('have.length.greaterThan', 0);
  });

  it('obsahuje správné sloupce', () => {
    const HEADERS = ['Jméno', 'Email', 'Telefon'];
    cy.get('table thead th').should('have.length', HEADERS.length);
  });

  it('zobrazí správná data v prvním řádku', () => {
    cy.dataCy('table-row')
      .first()
      .within(() => {
        cy.dataCy('cell-name').should('be.visible');
        cy.dataCy('cell-email').should('be.visible');
      });
  });

  it('umožní kliknout na detail', () => {
    cy.intercept('GET', '**/contact/detail/**').as('getDetail');
    cy.dataCy('table-row')
      .first()
      .within(() => {
        cy.dataCy('detail-button').click();
      });
    cy.wait('@getDetail');
    cy.url().should('include', '/contact/detail');
  });
});
