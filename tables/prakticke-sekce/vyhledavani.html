Testování vyhledávání

Kompletní průvodce testováním vyhledávacích funkcí v Cypressu. Zahrnuje vyhledávání v tabulkách, globální vyhledávání a ověření výsledků.

Základní vyhledávání

Vyhledávání pomocí custom příkazu

// Definice custom příkazu
Cypress.Commands.add('searchInput', (dataCy, typeValue, { clear, labelValue } = {}) => {
  cy.dataCy(dataCy)
    .scrollIntoView()
    .within(() => {
      const input = cy.get('input');
      
      if (labelValue) {
        cy.get('se-input-label').should('contain.text', labelValue);
      }
      
      if (clear) {
        input.clear();
      }
      
      input.type(typeValue, { delay: 0 });
    });
});

// Použití
cy.intercept({ method: 'GET', url: '**/search/global**' }).as('search');
cy.searchInput('search-container', 'Hledaný text');
cy.wait('@search').its('response.statusCode').should('eq', 200);

Vyhledávání s čekáním na výsledky

cy.intercept({ method: 'GET', url: '**/api/search**' }).as('search');
cy.searchInput('search-container', 'Hledaný text');
cy.wait('@search');
cy.dataCy('search-results').should('be.visible');

Ověření výsledků vyhledávání

Ověření zobrazení výsledků

cy.dataCy('search-results')
  .should('be.visible')
  .within(() => {
    cy.dataCy('result-item').should('have.length.greaterThan', 0);
  });

Ověření obsahu výsledků

cy.dataCy('result-item')
  .should('contain.text', 'Hledaný text')
  .and('be.visible');

Navigace z výsledků

cy.intercept({ method: 'GET', url: '**/contact/detail/**' }).as('goToDetail');
cy.dataCy('result-item')
  .first()
  .click();
cy.wait('@goToDetail');
cy.url().should('include', '/contact/detail');

Prázdné výsledky

cy.searchInput('search-container', 'Neexistující text');
cy.dataCy('no-results-message')
  .should('be.visible')
  .and('contain.text', 'Bohužel jsme nenašli žádné výsledky');

Vymazání vyhledávání

cy.dataCy('clear-button')
  .should('be.visible')
  .click();

cy.dataCy('search-results').should('not.exist');
cy.get('input').should('have.value', '');

Kompletní příklad

describe('Globální vyhledávání', () => {
  const SEARCH_TERM = 'Můj první seznam';

  before(() => {
    cy.tokenAuth('/dashboard');
  });

  it('vyhledá existující položku', () => {
    cy.intercept({ method: 'GET', url: '**/search/global**' }).as('search');
    cy.searchInput('search-container', SEARCH_TERM);
    cy.wait('@search').its('response.statusCode').should('eq', 200);
  });

  it('zobrazí výsledky vyhledávání', () => {
    cy.dataCy('search-results').should('be.visible');
    cy.dataCy('result-item')
      .should('be.visible')
      .and('contain.text', SEARCH_TERM);
  });

  it('naviguje na výsledek', () => {
    cy.intercept({ method: 'GET', url: '**/contact-list/**' }).as('goToItem');
    cy.dataCy('result-item').first().click();
    cy.wait('@goToItem');
    cy.dataCy('page-header').should('contain.text', SEARCH_TERM);
  });
});
